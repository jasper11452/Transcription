def mask(secret: str | None) -> str:
    """只暴露尾号，长度也只给粗粒度信息。"""
    if not secret:
        return "未配置"
    return f"已配置（尾号 {secret[-4:]}）"


def mask_preview(secret: str | None) -> str:
    """输入框占位符用的掩码：固定长度的点 + 尾号，中间一律掩盖。

    和 `mask` 的区别：`mask` 是"配没配"的状态文案（未配置时返回中文），这里是
    回填到输入框里的视觉占位符（未配置时返回空串）。两者都只暴露尾号 4 位。
    """
    if not secret:
        return ""
    return f"{'•' * 8}{secret[-4:]}"
